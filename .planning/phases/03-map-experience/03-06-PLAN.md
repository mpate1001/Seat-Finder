    // (mount → cleanup → mount). A naive cleanup that calls history.back()
    // triggers an async popstate that the second mount's listener catches —
    // onClose() fires, MapView unmounts, user sees "nothing happens".

    // follow-up mount runs first (StrictMode case), it flips a shared flag

      // supersedes that cleanup (we are in a StrictMode remount).
      h.pendingPop = false;

      if (!h.pushed) {
