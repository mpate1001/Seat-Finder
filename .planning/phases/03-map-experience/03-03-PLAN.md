```tsx
import { useEffect, useRef, useState } from 'react';
import {
  TransformWrapper,
  TransformComponent,
  type ReactZoomPanPinchRef,
} from 'react-zoom-pan-pinch';
import { Guest } from '../types';
import floorPlanConfig from '../config/floorPlan.json';
import FloorPlan from './FloorPlan';
import './MapView.css';

interface MapViewProps {
  guest: Guest;
  onClose: () => void;
}

const tablePositions = floorPlanConfig.tablePositions as Record<string, { x: number; y: number }>;

export default function MapView({ guest, onClose }: MapViewProps) {
  const transformRef = useRef<ReactZoomPanPinchRef | null>(null);
  const assignedPinRef = useRef<HTMLDivElement | null>(null);
  const [imageLoaded, setImageLoaded] = useState(false);

  const hasValidPosition = Boolean(tablePositions[guest.tableNumber]);

  // Escape-to-close — copied verbatim from TableModal.tsx lines 12-22
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        onClose();
      }
    };
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [onClose]);

  // Browser back-button integration — from RESEARCH.md Pattern 6
  // IMPORTANT: never call history.back() inside the popstate handler (infinite loop risk)
  useEffect(() => {
    history.pushState({ mapOpen: true }, '');

    function handlePopState() {
      onClose();
    }

    window.addEventListener('popstate', handlePopState);
    return () => {
      window.removeEventListener('popstate', handlePopState);
      // If MapView is closing via × or Escape (not Back), pop our pushed entry
      if (history.state?.mapOpen) {
        history.back();
      }
    };
  }, [onClose]);

  // Animation orchestration: 250ms hold → 700ms zoom to 2.75× on assigned pin
  // Only runs when image is loaded AND the guest has a valid position
  useEffect(() => {
    if (!imageLoaded || !hasValidPosition) return;

    const prefersReducedMotion =
      typeof window !== 'undefined' &&
      window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    const holdMs = prefersReducedMotion ? 0 : 250;
    const zoomMs = prefersReducedMotion ? 0 : 700;

    const timer = window.setTimeout(() => {
      if (!transformRef.current || !assignedPinRef.current) return;
      transformRef.current.zoomToElement(
        assignedPinRef.current,
        2.75,
        zoomMs,
        'easeOutQuart',
        0,
        64,
      );
    }, holdMs);

    return () => window.clearTimeout(timer);
  }, [imageLoaded, hasValidPosition, guest.tableNumber]);

  function handleImageLoad() {
    setImageLoaded(true);
  }

  return (
    <div className="map-overlay" role="dialog" aria-modal="true" aria-label="Floor plan map">
      <div className="map-overlay-card" aria-live="polite">
        <h2 className="map-overlay-card-greeting">
          Welcome, {guest.firstName}! — Table {guest.tableNumber}
        </h2>
        {guest.description && (
          <p className="map-overlay-card-description">{guest.description}</p>
        )}
        {!hasValidPosition && (
          <p className="map-overlay-card-fallback">
            Table {guest.tableNumber} — please ask staff for directions
          </p>
        )}
      </div>

      <button
        type="button"
        className="map-close-button"
        onClick={onClose}
        aria-label="Close map"
        title="Close map (Esc)"
      >
        &times;
      </button>

      <div className="map-surface">
        <TransformWrapper
          ref={transformRef}
          initialScale={1}
          minScale={0.5}
          maxScale={6}
          centerOnInit={true}
          limitToBounds={true}
          centerZoomedOut={true}
          smooth={true}
          wheel={{ step: 0.2 }}
          doubleClick={{ mode: 'toggle', step: 2.75 }}
          pinch={{ disabled: false }}
          panning={{ velocityDisabled: false }}
        >
          <TransformComponent
            wrapperClass="map-transform-wrapper"
            contentClass="map-transform-content"
          >
            <FloorPlan
              tableNumber={guest.tableNumber}
              assignedPinRef={assignedPinRef}
              onImageLoad={handleImageLoad}
            />
          </TransformComponent>
        </TransformWrapper>
      </div>
    </div>
  );
}
```

**Copy invariants (UI-SPEC Copywriting Contract — checker will enforce):**
- Em-dash `—` (U+2014) with surrounding spaces in the greeting line: `Welcome, {firstName}! — Table {N}`. NOT a hyphen, NOT an en-dash.
- Exclamation on the greeting, no exclamation on `Table {N}`.
- Fallback text is EXACTLY: `Table {N} — please ask staff for directions` (em-dash, leading word "Table", trailing word "directions").
- Close button: `aria-label="Close map"`, `title="Close map (Esc)"`, visible glyph `&times;`.

**Why these specific library props:**
- `minScale: 0.5` — allows slight pinch-out below fit-to-viewport so users can see "around" the floor plan; `limitToBounds: true` + `centerZoomedOut: true` snaps back to fit when they release.
- `maxScale: 6` — text on the floor plan stays sharp on retina at 6× (UI-SPEC Map Surface table).
- `doubleClick.step: 2.75` — matches the auto-zoom target so double-tap toggles to the same level.
- `wheel.step: 0.2` — desktop mouse-wheel increments (UI-SPEC Animations & Motion table).

**Do NOT:**
- Do NOT call `zoomToElement` in an effect without the `imageLoaded` guard — per RESEARCH.md Pitfall 1, the library needs the image dimensions before zoom math works.
- Do NOT call `history.back()` inside `handlePopState` — per RESEARCH.md Pitfall / Pattern 6 infinite-loop warning.
- Do NOT put `onClick={onClose}` on the `.map-overlay` wrapper — full-screen overlays should NOT dismiss on backdrop click (no backdrop visible — map fills viewport, D-04). The × button + Escape + Back are the dismissal affordances (D-03).
- Do NOT add a "Got it!" done button (existed in TableModal line 45, superseded by D-03's three-way dismissal).
- Do NOT add a first-open hint toast (UI-SPEC Copywriting Contract explicitly says OMITTED).
- Do NOT capture `history.state?.mapOpen` via a TS cast — it's already `unknown`/any-ish at the DOM level; use `as any` cast inline ONLY if TS strict mode flags it. Preferred workaround: `(history.state as { mapOpen?: boolean } | null)?.mapOpen`.

The TypeScript strict-mode `noUnusedLocals` flag will complain if you destructure something you don't use — the skeleton above consumes every local. If the compiler warns on `handlePopState`'s unused `e: PopStateEvent` param, delete the parameter entirely (the zero-arg form is valid for event listeners).

Per D-03 (carry-forward from decision): Users dismiss via × button, Escape key, OR browser/hardware back button. All three paths must invoke `onClose()` exactly once.

Per D-12 (markers are visual only this phase): MapView does NOT need any click handlers on markers — that's FloorPlan's job in plan 04, and FloorPlan will apply `pointer-events: none` there.
