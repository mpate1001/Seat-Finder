    let cvPromise: Promise<CvNamespace> | null = null;
    export function getCv(): Promise<CvNamespace> {
      if (cvPromise) return cvPromise;
      cvPromise = (async () => {
        const mod = cvModule as unknown;
        if (mod instanceof Promise) return (await mod) as CvNamespace;
        if ((mod as { Mat?: unknown }).Mat) return mod as CvNamespace;
        await new Promise<void>((resolve) => {

    5. `const circles = await detectCircles(canvas);` — DEFAULT_HOUGH auto-picked by detect.ts.
    6. Emit `{stage:'cropping', message:'Cropping circles...'}`.
    7. For each circle: crop square of `2*r` via `ctx.getImageData(floor(cx-r), floor(cy-r), ceil(2*r), ceil(2*r))`. Guard x/y ≥ 0 with Math.max(0, …). Collect ImageData[].
    8. Emit `{stage:'ocr', message:'Reading circle numbers...', done:0, total:crops.length}`.
    9. `const ocr = await recognizeCircles(crops, (done, total) => onProgress({stage:'ocr', message:'Reading circle numbers...', done, total}));`.
    10. Build DraftPin[]: for each (circle, ocrResult) → DraftPin {
          id: crypto.randomUUID(),
          x: circle.cx / bitmap.width,
          y: circle.cy / bitmap.height,

          tableNumber: ocr.text.replace(/[^0-9]/g, '') || null,  // strip non-digits; null if empty

          status: tableNumber === null ? 'needs-number'
