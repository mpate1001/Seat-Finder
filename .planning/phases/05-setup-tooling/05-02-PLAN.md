// src/setup/houghDefaults.ts
import type { HoughOpts } from './types';

export function DEFAULT_HOUGH(imageWidth: number): HoughOpts {
  return {
    dp: 1,
    minDist:   Math.max(1, Math.round(imageWidth * 0.03)),
    param1:    100,
    param2:    30,
    minRadius: Math.max(1, Math.round(imageWidth * 0.012)),
    maxRadius: Math.max(2, Math.round(imageWidth * 0.035)),
  };
}
