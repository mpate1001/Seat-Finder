        ```ts
          const entries = Object.entries(cfg.tablePositions)
            .map(([id, pos]) => `    ${JSON.stringify(id)}: { "x": ${pos.x}, "y": ${pos.y} }`)
            .join(',\n');
          return (
            `{\n` +
            `  "imageFileName": ${JSON.stringify(cfg.imageFileName)},\n` +
            `  "tablePositions": {\n` +
            `${entries}\n` +
            `  }\n` +
            `}\n`
          );
        }
        ```
